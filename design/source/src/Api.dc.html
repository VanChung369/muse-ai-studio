Bắt đầu nhanh: tạo một ảnh

curl http://192.168.1.20:8080/v1/images/generations \
  -H "Authorization: Bearer $MUSE2API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "model": "muse-image",
    "prompt": "Ly cà phê sữa đá trên bậu cửa sổ, nắng sớm",
    "n": 1
  }'
 
import os
from openai import OpenAI

client = OpenAI(
    base_url="http://192.168.1.20:8080/v1",
    api_key=os.environ["MUSE2API_KEY"],
)

img = client.images.generate(
    model="muse-image",
    prompt="Ly cà phê sữa đá trên bậu cửa sổ, nắng sớm",
    n=1,
)
print(img.data[0].url)
 
import OpenAI from "openai";

const client = new OpenAI({
  baseURL: "http://192.168.1.20:8080/v1",
  apiKey: process.env.MUSE2API_KEY,
});

const img = await client.images.generate({
  model: "muse-image",
  prompt: "Ly cà phê sữa đá trên bậu cửa sổ, nắng sớm",
  n: 1,
});
console.log(img.data[0].url);
